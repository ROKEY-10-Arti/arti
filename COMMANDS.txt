# arti 실행 명령 모음 — 한 줄씩 그대로 복사해서 터미널에 붙여넣기
# (리눅스 터미널 붙여넣기: Ctrl+Shift+V)
# 지도: maps/gallery.yaml (두 로봇 공용, 저장소 폴더 안에서는 $ARTI_MAP)
# follow 로봇 = /robot1


# ===== 0. 처음 한 번 / 코드 받은 뒤 =====

cd ~/project/arti
pb


# ===== 1. robot1 주행 띄우기 (터미널 3개, 순서대로) =====

# [터미널 1] localization — "Creating bond timer..." 나오면 다음
ros2 launch arti_bringup localization.launch.py namespace:=/robot1   # 시작 위치는 init_pose.yaml undock (자동)

# [터미널 2] rviz — 위치가 어긋났을 때만 상단 "2D Pose Estimate"로 다시 찍기
ros2 launch turtlebot4_viz view_navigation.launch.py namespace:=/robot1

# [터미널 3] nav2 — 초기 위치 찍은 뒤 실행
ros2 launch turtlebot4_navigation nav2.launch.py namespace:=/robot1


# ===== 2. 잘 떴는지 확인 =====

ros2 topic echo /robot1/amcl_pose --once
ros2 action list | grep navigate_to_pose
ros2 topic hz /robot1/scan


# ===== 3. 로봇 도킹 / 언도킹 (로봇이 움직임 — 주변 확인 후) =====

ros2 action send_goal /robot1/undock irobot_create_msgs/action/Undock "{}"
ros2 action send_goal /robot1/dock irobot_create_msgs/action/Dock "{}"


# ===== 3-1. 관람객 추종 주행 (visitor_follower) =====
# 로봇이 움직임 — 언도킹, 주변 확인 후. Nav2에 목표를 보내지 말 것(cmd_vel이 섞임). Ctrl+C 하면 정지 후 종료
# 관람객 좌표는 인식 모듈이 /robot1/target_person/point_map (PointStamped, map 좌표) 으로 보낸다
# localization(1번 터미널 1)이 떠 있어야 함 (map→odom 변환)

ros2 run arti_follow visitor_follower --ros-args -r __ns:=/robot1
# 처음엔 느리게: 뒤에 -p max_linear:=0.15 -p max_angular:=0.8
# 로봇 안 움직이게 로그만: 뒤에 -r cmd_vel:=cmd_vel_dryrun

# 인식 모듈 없이 시험: map 위 한 점에 관람객이 서 있다고 보냄 → 로봇이 그 점 0.8 m 앞까지 가서 멈춤
#   x, y는 rviz "Publish Point"로 찍거나 ros2 topic echo /robot1/clicked_point 로 확인한 map 좌표
ros2 topic pub -r 5 /robot1/target_person/point_map geometry_msgs/msg/PointStamped "{header: {frame_id: map}, point: {x: 1.0, y: 0.0}}"

# 확인
ros2 topic echo /robot1/follow/state


# ===== 3-2. 키오스크 대신 액션·서비스 보내기 (PC3 → PC1, 로봇이 움직임) =====
# 통합 실행: ros2 launch arti_follow tour.launch.py   (localization이 떠 있어야 함)
# 순서: ① prepare_tour 성공(registered) → ② 130 s 안에 start_tour. 끝내기 = send_goal 터미널에서 Ctrl+C (취소)

# 0) 도크 위인지 먼저 확인 (is_docked: true 여야 함, 아니면 undock_failed / not_docked)
ros2 topic echo /robot1/dock_status --once

# ① 호출·등록 (undock → 위치 잡기 → 키오스크 쪽 회전 → 관람객 등록)
ros2 action send_goal --feedback /robot1/arti/prepare_tour arti_interfaces/action/PrepareTour "{start: true}"

# ② 자유관람 시작 (artwork_ids는 빈 리스트)
ros2 action send_goal --feedback /robot1/arti/start_tour arti_interfaces/action/StartTour "{artwork_ids: []}"

# 작품 설명 조회 서비스 (PC3 서버 /arti/description, 네임스페이스 없음, 작품 번호 int)
ros2 service list | grep arti/description
ros2 service call /arti/description arti_interfaces/srv/Description "{artwork_id: 1}"

# ===== 4. 토픽이 안 보일 때 =====

echo $ROS_DOMAIN_ID $ROS_DISCOVERY_SERVER
ros2 daemon stop; ros2 daemon start
ros2 topic list | grep robot1


# ===== 5. robot4 (코스주행) — 필요할 때만 =====

ros2 launch arti_bringup localization.launch.py namespace:=/robot4
ros2 launch turtlebot4_viz view_navigation.launch.py namespace:=/robot4
ros2 launch turtlebot4_navigation nav2.launch.py namespace:=/robot4
